\section{Model}\label{sec:model}
\todo{prose from \texttt{docs/notes/2026-08-31-diseno-BM17.md} and master plan §5--§6.}
People $[n]$; person $i$ is healthy with probability $q_i$, infected with $p_i=1-q_i$, independently; utility $u_i\ge0$ collected once, when $i$ is accredited healthy. A test on a pool $T$, $|T|\le G$, returns $R(T)=$ number of infected in $T$. Budget $B$, hard on every path. A history $h$ is the sequence of $(T,R(T))$. \textbf{Laminar class $\Lam$:} every new pool is disjoint from all tested pools (fresh) or a strict subset of a residual atom. \textbf{Posterior-zero accreditation:} $i$ is accredited when $\Pr(i\text{ infected}\mid h)=0$. Values: $\OPT^{\Lam}(B)$ (laminar, augmented), $\OPT^{D}_{\rm aug}(B)$ (unrestricted pools, augmented), $\OPT^{D}_{\rm bin}(B)$ (non-augmented: tests return only whether $R(T)>0$), $\OPT^{S}_{\rm bin}(B)$ (non-augmented static). Comparators: $\OPT^{S}_{\rm bin}\le\OPT^{D}_{\rm bin}\le\OPT^{D}_{\rm aug}$ and $\OPT^{\Lam}\le\OPT^{D}_{\rm aug}$.

\begin{lemma}[Lemma A: posterior factorizes over atoms]\label{lem:A}
Under a laminar history the posterior of the infection profile factorizes over the residual atoms, each atom's factor being the prior restricted to the atom conditioned on its count; the count of every atom is determined by $h$. \status{DEMOSTRADO (i); (ii)--(iii) and coherence pending; \texttt{lemma\_A\_laminar\_inference.tex}}
\end{lemma}

\section{Separation: augmented laminar versus the classical model}\label{sec:sep}

\begin{lemma}[Classical bound in the rare-health regime]\label{lem:classical}
In the non-augmented model with homogeneous $q\le\tfrac12$ and $u\equiv1$, every dynamic policy with $B$ tests collects at most $Bq$ in expectation, and $B$ individual tests collect exactly $Bq$: $\OPT^{D}_{\rm bin}(B)=\OPT^{S}_{\rm bin}(B)=Bq$.
\end{lemma}
\begin{proof}[Proof sketch] \status{PENDIENTE: A writes it (28--29 Sep); = companion Thm 7.1, validated}
A test on $m$ not-yet-accredited people accredits them all iff it reads negative, so without history it yields $m\,q^m\le q$ for $q\le\tfrac12$ (equality at $m=1$, and at $m=2$ when $q=\tfrac12$). History cannot help: negative results accredit and remove their people; the remaining results are positive, i.e.\ decreasing events in the health indicators, so by Harris' inequality $\Pr(\text{all of }T\text{ healthy}\mid h)\le q^{m}$. Summing over $B$ tests gives $Bq$.
\end{proof}

\begin{proposition}[Cover-then-bisect lower bound]\label{prop:cbs}
For homogeneous $q$, $u\equiv1$, $n\ge kG$ with $k=B-\lceil\log_2 G\rceil\ge1$, under posterior-zero accreditation $\OPT^{\Lam}(B)\ge 1-(1-q)^{kG}$. Under strict accreditation the same plan gives $1-(1-q)^{(k-1)G}$. \status{DEMOSTRADO (master plan §16, arithmetic verified); the plan is not optimal (Map 3)}
\end{proposition}

\begin{theorem}[Unbounded separation]\label{thm:sep}
For $q\le\tfrac12$, $\displaystyle \frac{\OPT^{\Lam}(B)}{\OPT^{D}_{\rm bin}(B)}\ \ge\ \frac{1-(1-q)^{(B-\lceil\log_2G\rceil)G}}{Bq}$. Along $G=2^j$, $B=j+1$, $q=c/G$ the right-hand side is $\gtrsim (1-e^{-c})\,G/(c\,(\log_2G+1))\to\infty$. \status{follows from Lemma~\ref{lem:classical} + Proposition~\ref{prop:cbs}; DEMOSTRADO once the lemma is written}
\end{theorem}

\paragraph{Numbers at the anchor.} $(q,G,B)=(0.05,16,7)$, $n=48$: classical optimum $0.35$; laminar $\ge0.9147$ (CBS) and $\ge1.1233$ (exact optimum restricted to $G\le8$, Map~3): factors $2.61$ and $3.21$ against the true classical optimum. Frontier $(0.05,8,6)$, $n=24$: exact laminar $0.8654$ against $0.30$, factor $2.88$. \status{VERIFICADO; \texttt{results/mapa\_homogeneo\_3.csv}}

\begin{remark}[Certificate without computing the optimum]
For populations where Lemma~\ref{lem:classical} does not apply, the factor-2 adaptivity bound of the non-augmented model (Francisco, in preparation) gives $\OPT^{D}_{\rm bin}\le 2\,\OPT^{S}_{\rm bin}$: any augmented laminar policy worth more than twice the static non-overlapping optimum provably beats every non-augmented dynamic policy. \status{DERIVACION CONDICIONAL to that paper}
\end{remark}

\section{The convention gap: strict versus posterior-zero clearing}\label{sec:gap}
Full statement and proofs in \texttt{proposicion\_brecha\_convencion.tex} (to be merged here).
\begin{proposition}[Convention gap]
(1) $C_{\str}(h)\subseteq C_{\pz}(h)$ for every history, hence $\OPT_{\str}(B)\le\OPT_{\pz}(B)$. (2) Every posterior-zero policy is replicated under strict clearing with at most $B-1$ additional tests, one per deduced-clean piece: $\OPT_{\pz}(B)\le\OPT_{\str}(2B-1)$. (3) Sharpness: with $n\ge3$, $G=2$, $B=2$, $u\equiv1$, the pair-first policy is worth $2q+q(q^2+p^2)$ under posterior-zero and $2q+q^2(2q-1)$ under strict, against $2q$ for two singletons; the gap $pq$ comes from one branch. \status{(1),(3) DEMOSTRADO, VERIFICADO $n=4$; (2) DEMOSTRADO, cross-review by B pending}
\end{proposition}
\begin{corollary}
In the two-test game strict clearing reproduces the threshold $q=\tfrac12$ of the non-augmented model (singletons optimal below it); under posterior-zero there is no threshold. The convention flips the optimal first move. Francisco (2026-09-22) states the mechanism of (2) independently: ``the cost of going from soft to hard clearing is one more test per group, not necessarily laminar''.
\end{corollary}
